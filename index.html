async function toggleListening() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert("El reconocimiento de voz no está soportado en este navegador."); return; } if (isListening) { if (recognition) { try { recognition.stop(); } catch(e){} } stopListeningVisuals(); return; } // Forzar la liberación previa del hardware de audio en iOS try { const stream = await navigator.mediaDevices.getUserMedia({ audio: true }); stream.getTracks().forEach(track => track.stop()); } catch (err) { console.log("Permiso ya concedido o no requerido directamente:", err); } recognition = new SpeechRecognition(); recognition.lang = 'es-ES'; recognition.continuous = false; recognition.interimResults = false; recognition.onstart = () => { isListening = true; const micBtn = document.getElementById('micBtn'); const statusText = document.getElementById('statusText'); if(micBtn) micBtn.classList.add('listening'); if(statusText) { statusText.innerText = "Escuchando jugada..."; statusText.classList.add('text-orange-600'); } }; recognition.onresult = (event) => { const text = event.results[0][0].transcript.toLowerCase(); const transcriptEl = document.getElementById('transcript'); if(transcriptEl) transcriptEl.innerText = `"${text}"`; processVoiceCommand(text); }; recognition.onerror = () => { stopListeningVisuals(); }; recognition.onend = () => { stopListeningVisuals(); }; try { recognition.start(); } catch(e) { stopListeningVisuals(); } }
